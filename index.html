// --- CLOUD SYNC ENGINE (GOOGLE SHEETS) --- async function sendToCloud(payload) { if (!syncConfig.sheetUrl) return; const statusEl = document.getElementById('saveIndicator'); if (statusEl) { statusEl.innerHTML = 'Syncing...'; statusEl.classList.remove('opacity-0'); } try { const rawData = JSON.stringify(payload); await fetch(syncConfig.sheetUrl, { method: 'POST', redirect: 'follow', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: rawData }); if (statusEl) { statusEl.innerHTML = 'Saved'; setTimeout(() => statusEl.classList.add('opacity-0'), 2000); } updateSyncBadge(true); } catch (err) { console.warn('Cloud sync post warning:', err); updateSyncBadge(false); } } async function loadFromCloud() { if (!syncConfig.sheetUrl) return; updateSyncBadge(null); // Amber syncing indicator try { const response = await fetch(syncConfig.sheetUrl + (syncConfig.sheetUrl.includes('?') ? '&' : '?') + 't=' + Date.now(), { method: 'GET', redirect: 'follow' }); const text = await response.text(); if (!text || text === '{}' || text.trim() === '') { // Sheet is empty: push current local state to populate cell A1 immediately await sendToCloud(appState); updateSyncBadge(true); return; } const cloudData = JSON.parse(text); if (cloudData && typeof cloudData === 'object' && Object.keys(cloudData).length > 0) { appState = Object.assign(appState, cloudData); localStorage.setItem('taylor_finances_data', JSON.stringify(appState)); recalcCashFlowBalances(); recalcAll(); renderCategorizedExpenses(); renderIncomeTable(); renderHistoryTable(); updateSyncBadge(true); } } catch (err) { console.warn('Cloud sync load warning:', err); updateSyncBadge(false); } } function saveSyncConfig() { const input = document.getElementById('sheetUrlInput'); const url = input ? input.value.trim() : ''; syncConfig.sheetUrl = url; localStorage.setItem('taylor_sheet_url', url); closeSyncModal(); if (url) { updateSyncBadge(null); // Force an immediate push of the current state so the sheet populates cell A1 right away sendToCloud(appState).then(() => { loadFromCloud(); }); } else { updateSyncBadge(false); } }
